function setShards(amount) { const input = document.getElementById("shards"); input.value = amount; calculate(); } function calculate() { const input = document.getElementById("shards"); const price = document.getElementById("price"); if (!input || !price) { return; } let shards = Number(input.value); if (isNaN(shards) || shards < 0) { shards = 0; } const total = shards * 1.50; price.innerText = total.toFixed(2) + " ₽"; }
